//-> SAVE/LOAD GAME
function SaveScript(){
	var saveName = string(global.saveName)+".sav";
	
	if file_exists(saveName){
		file_delete(saveName);
	}
	
	ini_open(saveName);
	ini_write_real("PLAYER","xx",obj_Player.x);
	ini_write_real("PLAYER","yy",obj_Player.y);
	ini_write_real("PLAYER","vida",global.plrVida);
	ini_write_real("PLAYER","sala",room);
	ini_close();
}

function LoadScript(){
	var saveName = global.saveName+".sav";
	ini_open(saveName);
	obj_Player.x = ini_read_real("PLAYER","xx",obj_Player.x);
	obj_Player.y = ini_read_real("PLAYER","yy",obj_Player.y);
	global.plrVida = ini_read_real("PLAYER","vida",global.plrVida);
	room_goto(ini_read_real("PLAYER","sala",0));
	if instance_exists(obj_SaveLamp){ obj_SaveLamp.estate = false; }
	ini_close();
}

//SAVE/LOAD CONTROLS
function SaveControlScript(button,variavel){
	ini_open("controls");
	ini_write_string("CONTROL",button,constantToString(variavel));
	ini_close();
}

function LoadControlsScript(){
	if file_exists("controls"){
		ini_open("controls");
		global.kAndarEsq = ini_read_string("CONTROL","aEsq","A");
		global.kAndarDir = ini_read_string("CONTROL","aDir","D");
		global.kPular = ini_read_string("CONTROL","Pula","Espaço");
		global.kDash = ini_read_string("CONTROL","Dash","Ctrl");
		global.kAtaque1 = ini_read_string("CONTROL","Atq1","Q");
		global.kAtaque2 = ini_read_string("CONTROL","Atq2","R");
		ini_close();
	}else{
		ini_open("controls");
		ini_write_string("CONTROL","aEsq","A");
		ini_write_string("CONTROL","aDir","D");
		ini_write_string("CONTROL","Pula","Espaço");
		ini_write_string("CONTROL","Dash","Ctrl");
		ini_write_string("CONTROL","Atq1","Q");
		ini_write_string("CONTROL","Atq2","E");
		ini_close();
	}
}

function LoadControlButton(button){
	ini_open("controls");
	return ini_read_string("CONTROL",button,"NONE");
	ini_close();
}

function LoadButton(control){
	ini_open("controls");
	return stringToConstant(ini_read_string("CONTROL",control,"NONE"));
	ini_close();
}

//SAVE/LOAD SAVE NAMES AND VOLUME
function SaveSaveNames(SaveName){
	ini_open("SaveNames");
	var num = round(real(ini_read_string("QTDESAVES","num","0")));
	
	ini_write_string("SAVENAMES",string(num+1),SaveName);
	ini_write_string("QTDESAVES","num",string(num+1));
	ini_close();
}

function LoadSaveNames(idSave){
	ini_open("SaveNames");
	return ini_read_string("SAVENAMES",string(idSave),"NÂO HÁ JOGOS SALVOS");
	ini_close();
}

function LoadQtdeSaves(){
	ini_open("SaveNames");
	return round(real(ini_read_string("QTDESAVES","num","1")));
	ini_close();
}

function SaveVolume(){
	ini_open("SaveNames");
	ini_write_real("VOLUME","VOLUME",global.volume);
	ini_close();
}

function LoadVolume(){
	ini_open("SaveNames");
	return round(ini_read_real("VOLUME","VOLUME",50));
	ini_close();
}
